import { Account, bytes, uint64 } from '@algorandfoundation/algorand-typescript'

export type PoolInfo = {
  poolType: uint64
  assetA: uint64
  assetB: uint64
  escrow: uint64
  lpAsset: uint64
  reserveA: uint64
  reserveB: uint64
  lpSupply: uint64
  feeBps: uint64
  maxHookFeeBps: uint64
  hookPoints: uint64
  protocolFeesA: uint64
  protocolFeesB: uint64
  binStep: uint64
  activeId: uint64
  virtualA: uint64
  virtualB: uint64
  graduateAtB: uint64
  /** bins: Q64.64 price of the active bin, kept by the bins book */
  binPrice: bytes<16>
  externalA: uint64
  externalB: uint64
  creator: Account
  paused: boolean
}

export type HookConfig = { app: uint64; points: uint64; perms: uint64 }
export type PairKey = { assetA: uint64; assetB: uint64 }
export type PageKey = { poolId: uint64; page: uint64 }
export type PositionKey = { poolId: uint64; owner: Account; lowerId: uint64 }

export type ExternalPosition = {
  hook: uint64
  valuer: uint64
  side: uint64
  receipt: uint64
  capBps: uint64
  maxMarkupBpsPerDay: uint64
  booked: uint64
  rate: uint64
  lastValued: uint64
}

export type HookContext = {
  point: uint64
  poolId: uint64
  poolType: uint64
  escrow: Account
  sender: Account
  trader: Account
  recipient: Account
  assetIn: uint64
  assetOut: uint64
  amountIn: uint64
  amountOut: uint64
  reserveIn: uint64
  reserveOut: uint64
  lpAmount: uint64
  lpSupply: uint64
  feeBps: uint64
  binStep: uint64
  activeId: uint64
  startPriceX64: bytes<16>
  priceX64: bytes<16>
  data: bytes
}

export type HookResult = { flags: uint64; feeBps: uint64; hookFee: uint64 }
export type SwapResult = { assetOut: uint64; amountOut: uint64 }
export type SwapEvent = { poolId: uint64; trader: Account; assetIn: uint64; amountIn: uint64; assetOut: uint64; amountOut: uint64; feeBps: uint64; hookFees: uint64; reserveA: uint64; reserveB: uint64; activeId: uint64 }
export type LiquidityEvent = { poolId: uint64; caller: Account; amountA: uint64; amountB: uint64; lpAmount: uint64; reserveA: uint64; reserveB: uint64 }
export type CustodyEvent = { poolId: uint64; hook: uint64; point: uint64; reserveA: uint64; reserveB: uint64 }
export type BinSwapResult = { amountOut: uint64; protocolFee: uint64; activeId: uint64; price: bytes<16> }
export type BinLiquidityResult = { amountA: uint64; amountB: uint64; price: bytes<16> }
