  // PRIVATE: ACCESS & STATE ----------------------------------------------------------------------

  private requireAdmin(): void {
    assert(Txn.sender === this.admin.value, errAdminOnly)
  }

  private loadPool(poolId: uint64): PoolInfo {
    assert(this.pools(poolId).exists, errPoolNotFound)
    return clone(this.pools(poolId).value)
  }

  private escrowOf(pool: PoolInfo): Account {
    return Application(pool.escrow).address
  }

  private resolveTrader(trader: Account): Account {
    if (trader === Global.zeroAddress || trader === Txn.sender) {
      return Txn.sender
    }
    assert(this.approvedForwarders(Global.callerApplicationId).exists, errForwarder)
    return trader
  }

  // PRIVATE: TRANSFERS ---------------------------------------------------------------------------

  private holding(account: Account, asset: uint64): uint64 {
    if (asset === 0) {
      return account.balance - account.minBalance
    }
    const [balance, optedIn] = op.AssetHolding.assetBalance(account, asset)
    return optedIn ? balance : 0
  }

  private send(sender: Account, receiver: Account, asset: uint64, amount: uint64): void {
    if (asset === 0) {
      itxn.payment({ sender, receiver, amount }).submit()
    } else {
      itxn.assetTransfer({ sender, assetReceiver: receiver, xferAsset: asset, assetAmount: amount }).submit()
    }
  }

  private readDeposit(txn: gtxn.Transaction, receiver: Account): Deposit {
    assert(txn.rekeyTo === Global.zeroAddress, errInvalidInput)
    if (txn.type === TransactionType.Payment) {
      assert(txn.receiver === receiver, errInvalidInput)
      assert(txn.closeRemainderTo === Global.zeroAddress, errInvalidInput)
      return { asset: 0, amount: txn.amount }
    }
    assert(txn.type === TransactionType.AssetTransfer, errInvalidInput)
    assert(txn.assetReceiver === receiver, errInvalidInput)
    assert(txn.assetCloseTo === Global.zeroAddress, errInvalidInput)
    assert(txn.assetSender === Global.zeroAddress, errInvalidInput)
    return { asset: txn.xferAsset.id, amount: txn.assetAmount }
  }

  // PRIVATE: MATH --------------------------------------------------------------------------------

  private mulDiv(a: uint64, b: uint64, c: uint64): uint64 {
    return op.divw(...op.mulw(a, b), c)
  }

  private mulDivUp(a: uint64, b: uint64, c: uint64): uint64 {
    const [qHi, qLo, rHi, rLo] = op.divmodw(...op.mulw(a, b), 0, c)
    assert(qHi === 0, errInvalidInput)
    return rHi === 0 && rLo === 0 ? qLo : qLo + 1
  }

  private sqrtProduct(a: uint64, b: uint64): uint64 {
    return op.btoi(Bytes(op.bsqrt(BigUint(a) * BigUint(b))))
  }

  private b16(v: biguint): bytes<16> {
    return op.bzero(16).bitwiseOr(Bytes(v)).slice(-16).toFixed({ length: 16 })
  }

  /** Q64.64 price of A in B for a ratio (b / a) */
  private ratioPrice(a: uint64, b: uint64): bytes<16> {
    if (a === 0) return op.bzero(16).toFixed({ length: 16 })
    const [qHi, qLo, rHi, rLo] = op.divmodw(b, 0, 0, a)
    return op.itob(qHi).concat(op.itob(qLo)).toFixed({ length: 16 })
  }

  private spotPrice(pool: PoolInfo): bytes<16> {
    if (pool.poolType === PoolTypeBins) {
      return pool.binPrice
    }
    return this.ratioPrice(pool.reserveA + pool.virtualA, pool.reserveB + pool.virtualB)
  }

  private cpSwap(pool: PoolInfo, inIsA: boolean, amountIn: uint64, feeBps: uint64): SwapMath {
    const reserveIn: uint64 = inIsA ? pool.reserveA + pool.virtualA : pool.reserveB + pool.virtualB
    const reserveOut: uint64 = inIsA ? pool.reserveB + pool.virtualB : pool.reserveA + pool.virtualA
    const liquidOut: uint64 = inIsA ? pool.reserveB - pool.externalB : pool.reserveA - pool.externalA
    assert(reserveIn > 0 && reserveOut > 0, errInsufficientLiquidity)
    const lpFee = this.mulDiv(amountIn, feeBps, BpsDivisor)
    const protocolFee = this.mulDiv(lpFee, this.protocolFeeBps.value, BpsDivisor)
    const effectiveIn: uint64 = amountIn - lpFee
    const amountOut = this.mulDiv(effectiveIn, reserveOut, reserveIn + effectiveIn)
    assert(amountOut > 0 && amountOut < liquidOut, errInsufficientLiquidity)
    return { amountOut, protocolFee, reserveInDelta: amountIn - protocolFee, activeId: 0, price: pool.binPrice }
  }

  /** Pool-type dispatch: the only place swap math branches on poolType */
  private curveSwap(poolId: uint64, pool: PoolInfo, inIsA: boolean, amountIn: uint64, feeBps: uint64, commit: boolean): SwapMath {
    assert(amountIn > 0, errInvalidInput)
    if (pool.poolType === PoolTypeBins) {
      const r = abiCall<typeof PluginDexBins.prototype.swap>({
        appId: this.bins.value,
        args: [poolId, pool.binStep, pool.activeId, inIsA, amountIn, feeBps, feeBps === 0 ? 0 : this.protocolFeeBps.value, commit],
      }).returnValue
      return { amountOut: r.amountOut, protocolFee: r.protocolFee, reserveInDelta: amountIn - r.protocolFee, activeId: r.activeId, price: r.price }
    }
    return this.cpSwap(pool, inIsA, amountIn, feeBps)
  }

  /** Apply a swap result to the aggregate pool record */
  private applySwap(pool: PoolInfo, inIsA: boolean, math: SwapMath): void {
    if (inIsA) {
      pool.reserveA += math.reserveInDelta
      pool.reserveB -= math.amountOut
      pool.protocolFeesA += math.protocolFee
    } else {
      pool.reserveB += math.reserveInDelta
      pool.reserveA -= math.amountOut
      pool.protocolFeesB += math.protocolFee
    }
    if (pool.poolType === PoolTypeBins) {
      pool.activeId = math.activeId
      pool.binPrice = math.price
    }
  }

  /** Credit value to LPs: CP reserves, or the active bin */
  private donate(poolId: uint64, pool: PoolInfo, amountA: uint64, amountB: uint64): void {
    pool.reserveA += amountA
    pool.reserveB += amountB
    if (pool.poolType === PoolTypeBins) {
      abiCall<typeof PluginDexBins.prototype.donate>({ appId: this.bins.value, args: [poolId, pool.activeId, amountA, amountB] })
    }
  }

  // PRIVATE: CUSTODY GUARD -----------------------------------------------------------------------

  /**
   * Generalized custody guard. Book a custody hook's net change to effective reserves
   * (liquid + external) as a zero-fee swap against the pool's own curve:
   *  - gains only: donated to LPs
   *  - one side lost: the gained side is swapped through the curve at zero fee and must
   *    buy at least what was taken; any surplus is donated
   *  - both sides lost: revert
   * For CP this is exactly "k must not decrease"; for bins it is the bin path.
   */
  private bookCustody(poolId: uint64, pool: PoolInfo, gainA: uint64, lossA: uint64, gainB: uint64, lossB: uint64): void {
    if (lossA === 0 && lossB === 0) {
      this.donate(poolId, pool, gainA, gainB)
      return
    }
    assert(lossA === 0 || lossB === 0, errCustodyInvariant)
    const inIsA = lossB > 0
    const gain: uint64 = inIsA ? gainA : gainB
    const loss: uint64 = inIsA ? lossB : lossA
    assert(gain > 0, errCustodyInvariant)
    const math = this.curveSwap(poolId, pool, inIsA, gain, 0, true)
    assert(math.amountOut >= loss, errCustodyInvariant)
    this.applySwap(pool, inIsA, math)
    const surplus: uint64 = math.amountOut - loss
    this.donate(poolId, pool, inIsA ? 0 : surplus, inIsA ? surplus : 0)
  }

  /** Revalue the pool's external positions managed by `hook` (0 = all); returns signed deltas via pool */
  private revalueExternals(poolId: uint64, pool: PoolInfo, hook: uint64): void {
    if (!this.externals(poolId).exists) return
    const positions = clone(this.externals(poolId).value)
    const escrow = this.escrowOf(pool)
    for (let i: uint64 = 0; i < positions.length; i += 1) {
      const p = clone(positions[i])
      if (hook !== 0 && p.hook !== hook) continue
      const units: uint64 = p.receipt === 0 ? 0 : this.holding(escrow, p.receipt)
      let value = abiCall<typeof ExternalValuer.prototype.valueOf>({
        appId: p.valuer,
        args: [escrow, p.receipt, units, p.side === 0 ? pool.assetA : pool.assetB],
      }).returnValue
      // rate limit markups: value per receipt unit may rise at most maxMarkupBpsPerDay
      const rate: uint64 = units === 0 ? 0 : this.mulDiv(value, 1_000_000_000_000, units)
      if (p.rate > 0 && rate > p.rate) {
        const elapsed: uint64 = Global.latestTimestamp - p.lastValued
        const maxRate: uint64 = p.rate + this.mulDiv(p.rate, p.maxMarkupBpsPerDay * elapsed, BpsDivisor * 86_400)
        if (rate > maxRate) value = this.mulDiv(units, maxRate, 1_000_000_000_000)
      }
      const reserve: uint64 = p.side === 0 ? pool.reserveA : pool.reserveB
      assert(value <= this.mulDiv(reserve - p.booked + value, p.capBps, BpsDivisor), errExternalCap)
      if (p.side === 0) {
        pool.externalA = pool.externalA - p.booked + value
        pool.reserveA = pool.reserveA - p.booked + value
      } else {
        pool.externalB = pool.externalB - p.booked + value
        pool.reserveB = pool.reserveB - p.booked + value
      }
      p.booked = value
      p.rate = rate
      p.lastValued = Global.latestTimestamp
      positions[i] = clone(p)
    }
    this.externals(poolId).value = clone(positions)
  }

  // PRIVATE: HOOKS -------------------------------------------------------------------------------

  private runHooks(poolId: uint64, pool: PoolInfo, point: uint64, base: HookContext, inIsA: boolean): HookRun {
    const run: HookRun = { feeBps: base.feeBps, hookFees: 0, graduate: false }
    if ((pool.hookPoints & point) === 0) {
      return run
    }

    const escrow = this.escrowOf(pool)
    const hooks = clone(this.poolHooks(poolId).value)
    const maxHookFees = this.mulDiv(base.amountIn, pool.maxHookFeeBps, BpsDivisor)

    for (let i: uint64 = 0; i < hooks.length; i += 1) {
      const hook = clone(hooks[i])
      if ((hook.points & point) === 0) {
        continue
      }

      const ctx = clone(base)
      ctx.point = point
      ctx.feeBps = run.feeBps
      ctx.reserveIn = inIsA ? pool.reserveA + pool.virtualA : pool.reserveB + pool.virtualB
      ctx.reserveOut = inIsA ? pool.reserveB + pool.virtualB : pool.reserveA + pool.virtualA
      ctx.activeId = pool.activeId
      ctx.priceX64 = this.spotPrice(pool)

      const hookApp = Application(hook.app)
      const custody = (hook.perms & HookPermCustody) !== 0 && point !== HookPointQuote

      let balanceA: uint64 = 0
      let balanceB: uint64 = 0
      let extA: uint64 = 0
      let extB: uint64 = 0
      if (custody) {
        balanceA = this.holding(escrow, pool.assetA)
        balanceB = this.holding(escrow, pool.assetB)
        extA = pool.externalA
        extB = pool.externalB
        itxn.payment({ sender: escrow, receiver: escrow, amount: 0, rekeyTo: hookApp.address }).submit()
      }

      const result = abiCall<typeof PluginDexHook.prototype.akitaDexHook>({
        appId: hookApp,
        args: [Global.currentApplicationId, i, ctx],
      }).returnValue

      if (custody) {
        assert(escrow.authAddress === Global.currentApplicationAddress, errMissingRekeyBack)
        // external positions this hook manages are re-marked; their change in value
        // counts like a balance change of the side they belong to
        if ((hook.perms & HookPermExternal) !== 0) {
          pool.reserveA -= pool.externalA - extA
          pool.reserveB -= pool.externalB - extB
          this.revalueExternals(poolId, pool, hook.app)
        }
        const afterA: uint64 = this.holding(escrow, pool.assetA) + pool.externalA
        const afterB: uint64 = this.holding(escrow, pool.assetB) + pool.externalB
        const beforeA: uint64 = balanceA + extA
        const beforeB: uint64 = balanceB + extB
        pool.reserveA -= pool.externalA - extA
        pool.reserveB -= pool.externalB - extB
        this.bookCustody(
          poolId,
          pool,
          afterA > beforeA ? afterA - beforeA : 0,
          beforeA > afterA ? beforeA - afterA : 0,
          afterB > beforeB ? afterB - beforeB : 0,
          beforeB > afterB ? beforeB - afterB : 0,
        )
        emit<CustodyEvent>({ poolId, hook: hook.app, point, reserveA: pool.reserveA, reserveB: pool.reserveB })
      }

      const pricingPoint = point === HookPointBeforeSwap || point === HookPointQuote

      if ((result.flags & HookResultSetFee) !== 0) {
        assert(pricingPoint && (hook.perms & HookPermDynamicFee) !== 0, errHookPermission)
        assert(result.feeBps <= MaxFeeBps, errInvalidFee)
        run.feeBps = result.feeBps
      }
      if ((result.flags & HookResultGraduate) !== 0) {
        run.graduate = true
      }

      if (result.hookFee > 0) {
        assert(pricingPoint && (hook.perms & HookPermHookFee) !== 0, errHookPermission)
        run.hookFees += result.hookFee
        assert(run.hookFees <= maxHookFees, errHookFeeTooHigh)
        if (point === HookPointBeforeSwap) {
          this.send(escrow, hookApp.address, base.assetIn, result.hookFee)
        }
      }
    }

    return run
  }

  private context(
    poolId: uint64,
    pool: PoolInfo,
    trader: Account,
    recipient: Account,
    assetIn: uint64,
    assetOut: uint64,
    amountIn: uint64,
    amountOut: uint64,
    lpAmount: uint64,
    data: bytes,
  ): HookContext {
    const price = this.spotPrice(pool)
    return {
      point: 0,
      poolId,
      poolType: pool.poolType,
      escrow: this.escrowOf(pool),
      sender: Txn.sender,
      trader,
      recipient,
      assetIn,
      assetOut,
      amountIn,
      amountOut,
      reserveIn: 0,
      reserveOut: 0,
      lpAmount,
      lpSupply: pool.lpSupply,
      feeBps: pool.feeBps,
      binStep: pool.binStep,
      activeId: pool.activeId,
      startPriceX64: price,
      priceX64: price,
      data,
    }
  }

  // PRIVATE: SWAPS -------------------------------------------------------------------------------

  private swapCore(poolId: uint64, assetIn: uint64, amountIn: uint64, trader: Account, recipient: Account, data: bytes): SwapResult {
    const pool = this.loadPool(poolId)
    assert(!pool.paused, errPoolPaused)

    const inIsA = assetIn === pool.assetA
    assert(inIsA || assetIn === pool.assetB, errAssetNotInPool)
    const assetOut = inIsA ? pool.assetB : pool.assetA

    const ctx = this.context(poolId, pool, trader, recipient, assetIn, assetOut, amountIn, 0, 0, data)

    const before = this.runHooks(poolId, pool, HookPointBeforeSwap, ctx, inIsA)
    const netIn: uint64 = amountIn - before.hookFees
    const math = this.curveSwap(poolId, pool, inIsA, netIn, before.feeBps, true)
    this.applySwap(pool, inIsA, math)

    ctx.amountOut = math.amountOut
    ctx.feeBps = before.feeBps
    const after = this.runHooks(poolId, pool, HookPointAfterSwap, ctx, inIsA)

    if (pool.poolType === PoolTypeLaunch && (after.graduate || pool.reserveB >= pool.graduateAtB)) {
      this.graduate(poolId, pool)
    }

    this.pools(poolId).value = clone(pool)

    emit<SwapEvent>({
      poolId,
      trader,
      assetIn,
      amountIn,
      assetOut,
      amountOut: math.amountOut,
      feeBps: before.feeBps,
      hookFees: before.hookFees,
      reserveA: pool.reserveA,
      reserveB: pool.reserveB,
      activeId: pool.activeId,
    })

    return { assetOut, amountOut: math.amountOut }
  }

  /**
   * Launch -> CP: keep the curve's current price, seed a CP pool from the real B
   * reserve and the matching amount of A, send the rest of A to the creator's
   * configured sink and lock the minted LP in the DEX.
   */
  private graduate(poolId: uint64, pool: PoolInfo): void {
    const keepA = this.mulDiv(pool.reserveB, pool.reserveA + pool.virtualA, pool.reserveB + pool.virtualB)
    assert(keepA <= pool.reserveA, errGraduate)
    const excessA: uint64 = pool.reserveA - keepA
    pool.reserveA = keepA
    pool.virtualA = 0
    pool.virtualB = 0
    pool.poolType = PoolTypeConstantProduct
    pool.lpSupply = this.sqrtProduct(keepA, pool.reserveB)
    if (excessA > 0) this.send(this.escrowOf(pool), pool.creator, pool.assetA, excessA)
  }

  private quoteCore(poolId: uint64, assetIn: uint64, amountIn: uint64, trader: Account, data: bytes): SwapResult {
    const pool = this.loadPool(poolId)
    assert(!pool.paused, errPoolPaused)

    const inIsA = assetIn === pool.assetA
    assert(inIsA || assetIn === pool.assetB, errAssetNotInPool)
    const assetOut = inIsA ? pool.assetB : pool.assetA

    const ctx = this.context(poolId, pool, trader, trader, assetIn, assetOut, amountIn, 0, 0, data)
    const run = this.runHooks(poolId, pool, HookPointQuote, ctx, inIsA)
    const math = this.curveSwap(poolId, pool, inIsA, amountIn - run.hookFees, run.feeBps, false)
    return { assetOut, amountOut: math.amountOut }
  }

  private resolveRecipient(recipient: Account): Account {
    return recipient === Global.zeroAddress ? Txn.sender : recipient
  }

  // PRIVATE: LIQUIDITY ---------------------------------------------------------------------------

  private removeCore(
    lpTransfer: gtxn.AssetTransferTxn,
    poolId: uint64,
    minAmountA: uint64,
    minAmountB: uint64,
    data: bytes,
    withHooks: boolean,
  ): RemoveResult {
    const pool = this.loadPool(poolId)
    assert(pool.poolType !== PoolTypeBins, errPoolType)

    assert(lpTransfer.xferAsset.id === pool.lpAsset, errInvalidInput)
    assert(lpTransfer.assetReceiver === Global.currentApplicationAddress, errInvalidInput)
    assert(lpTransfer.assetCloseTo === Global.zeroAddress, errInvalidInput)
    assert(lpTransfer.assetSender === Global.zeroAddress, errInvalidInput)
    assert(lpTransfer.rekeyTo === Global.zeroAddress, errInvalidInput)
    const lpAmount = lpTransfer.assetAmount
    assert(lpAmount > 0, errInvalidInput)

    const ctx = this.context(poolId, pool, Txn.sender, Txn.sender, pool.assetA, pool.assetB, 0, 0, lpAmount, data)
    if (withHooks) this.runHooks(poolId, pool, HookPointBeforeRemoveLiquidity, ctx, true)

    // liquid part pro rata; external part in kind (receipt tokens) pro rata
    const amountA = this.mulDiv(lpAmount, pool.reserveA - pool.externalA, pool.lpSupply)
    const amountB = this.mulDiv(lpAmount, pool.reserveB - pool.externalB, pool.lpSupply)
    assert(amountA >= minAmountA && amountB >= minAmountB, errSlippage)

    const escrow = this.escrowOf(pool)
    if (this.externals(poolId).exists) {
      const positions = clone(this.externals(poolId).value)
      for (let i: uint64 = 0; i < positions.length; i += 1) {
        const p = clone(positions[i])
        const share = this.mulDiv(lpAmount, p.booked, pool.lpSupply)
        if (p.receipt !== 0) {
          const units = this.mulDiv(lpAmount, this.holding(escrow, p.receipt), pool.lpSupply)
          if (units > 0) this.send(escrow, Txn.sender, p.receipt, units)
        }
        p.booked -= share
        if (p.side === 0) {
          pool.externalA -= share
          pool.reserveA -= share
        } else {
          pool.externalB -= share
          pool.reserveB -= share
        }
        positions[i] = clone(p)
      }
      this.externals(poolId).value = clone(positions)
    }

    pool.reserveA -= amountA
    pool.reserveB -= amountB
    pool.lpSupply -= lpAmount

    if (withHooks) {
      ctx.amountIn = amountA
      ctx.amountOut = amountB
      ctx.lpSupply = pool.lpSupply
      this.runHooks(poolId, pool, HookPointAfterRemoveLiquidity, ctx, true)
    }

    this.pools(poolId).value = clone(pool)

    if (amountA > 0) this.send(escrow, Txn.sender, pool.assetA, amountA)
    if (amountB > 0) this.send(escrow, Txn.sender, pool.assetB, amountB)

    emit<LiquidityEvent>({ poolId, caller: Txn.sender, amountA, amountB, lpAmount, reserveA: pool.reserveA, reserveB: pool.reserveB })
    return { amountA, amountB }
  }

    this.requireAdmin()
    this.admin.value = admin
  }

  setProtocolFee(protocolFeeBps: uint64, treasury: Account): void {
    this.requireAdmin()
    assert(protocolFeeBps <= MaxProtocolFeeBps, errInvalidFee)
    this.protocolFeeBps.value = protocolFeeBps
    this.treasury.value = treasury
  }

  setForwarderApproval(forwarder: uint64, approved: boolean): void {
    this.requireAdmin()
    if (!approved) {
      if (this.approvedForwarders(forwarder).exists) this.approvedForwarders(forwarder).delete()
      return
    }
    this.approvedForwarders(forwarder).value = 1
  }

  setPoolPaused(poolId: uint64, paused: boolean): void {
    this.requireAdmin()
    assert(this.pools(poolId).exists, errPoolNotFound)
    this.pools(poolId).value.paused = paused
  }

  collectProtocolFees(poolId: uint64): void {
    const pool = this.loadPool(poolId)
    const escrow = this.escrowOf(pool)
    const feesA = pool.protocolFeesA
    const feesB = pool.protocolFeesB
    this.pools(poolId).value.protocolFeesA = 0
    this.pools(poolId).value.protocolFeesB = 0
    if (feesA > 0) this.send(escrow, this.treasury.value, pool.assetA, feesA)
    if (feesB > 0) this.send(escrow, this.treasury.value, pool.assetB, feesB)
  }

  /** Anyone: re-mark external positions (yield accrues to LPs) */
  revalue(poolId: uint64): void {
    const pool = this.loadPool(poolId)
    this.revalueExternals(poolId, pool, 0)
    this.pools(poolId).value = clone(pool)
  }

  /** Anyone: book unaccounted escrow surplus (donations, consensus rewards) to LPs */
  sync(poolId: uint64): void {
    const pool = this.loadPool(poolId)
    const escrow = this.escrowOf(pool)
    const liquidA: uint64 = this.holding(escrow, pool.assetA)
    const liquidB: uint64 = this.holding(escrow, pool.assetB)
    const bookedA: uint64 = pool.reserveA - pool.externalA + pool.protocolFeesA
    const bookedB: uint64 = pool.reserveB - pool.externalB + pool.protocolFeesB
    this.donate(poolId, pool, liquidA > bookedA ? liquidA - bookedA : 0, liquidB > bookedB ? liquidB - bookedB : 0)
    this.pools(poolId).value = clone(pool)
  }

  /**
   * Factory-only: adopt a pool whose escrow the factory created, funded, opted in and
   * rekeyed to this app. The factory has already validated hooks, valuers and params.
   */
  registerPool(
    payment: gtxn.PaymentTxn,
    poolType: uint64,
    assetA: uint64,
    assetB: uint64,
    escrowApp: Application,
    feeBps: uint64,
    maxHookFeeBps: uint64,
    hookPoints: uint64,
    curveParams: uint64[],
    creator: Account,
    hooks: HookConfig[],
    externals: ExternalPosition[],
    binPrice: bytes<16>,
  ): uint64 {
    assert(Global.callerApplicationId === this.factory.value.id, errAdminOnly)
    assert(payment.receiver === Global.currentApplicationAddress, errInvalidPayment)
    assert(escrowApp.address.authAddress === Global.currentApplicationAddress, errInvalidPool)
    const mbrBefore = Global.currentApplicationAddress.minBalance

    const poolId = this.poolCursor.value
    this.poolCursor.value = poolId + 1

    let lpAsset: uint64 = 0
    if (poolType !== PoolTypeBins) {
      lpAsset = itxn
        .assetConfig({
          total: LPAssetTotal,
          decimals: LPAssetDecimals,
          unitName: LPAssetUnitName,
          assetName: LPAssetName,
          reserve: Global.currentApplicationAddress,
        })
        .submit().createdAsset.id
    }

    const pool: PoolInfo = {
      poolType,
      assetA,
      assetB,
      escrow: escrowApp.id,
      lpAsset,
      reserveA: 0,
      reserveB: 0,
      lpSupply: 0,
      feeBps,
      maxHookFeeBps,
      hookPoints,
      protocolFeesA: 0,
      protocolFeesB: 0,
      binStep: poolType === PoolTypeBins ? curveParams[0] : 0,
      activeId: poolType === PoolTypeBins ? curveParams[1] : 0,
      virtualA: poolType === PoolTypeLaunch ? curveParams[0] : 0,
      virtualB: poolType === PoolTypeLaunch ? curveParams[1] : 0,
      graduateAtB: poolType === PoolTypeLaunch ? curveParams[2] : 0,
      binPrice: binPrice,
      externalA: 0,
      externalB: 0,
      creator,
      paused: false,
    }
    if (poolType === PoolTypeBins) {
      pool.binPrice = curveParams[2] === 0 ? pool.binPrice : pool.binPrice
    }
    if (hooks.length > 0) this.poolHooks(poolId).value = clone(hooks)
    if (externals.length > 0) this.externals(poolId).value = clone(externals)

    const ctx = this.context(poolId, pool, creator, creator, assetA, assetB, 0, 0, 0, Bytes(''))
    this.runHooks(poolId, pool, HookPointAfterInitialize, ctx, true)
    this.pools(poolId).value = clone(pool)

    const required: uint64 = Global.currentApplicationAddress.minBalance - mbrBefore
    assert(payment.amount >= required, errInvalidPayment)
    if (payment.amount > required) {
      itxn.payment({ receiver: payment.sender, amount: payment.amount - required }).submit()
    }
    return poolId
  }

  addLiquidity(depositA: gtxn.Transaction, depositB: gtxn.Transaction, poolId: uint64, minLpOut: uint64, data: bytes): uint64 {
    const pool = this.loadPool(poolId)
    assert(!pool.paused, errPoolPaused)
    assert(pool.poolType === PoolTypeConstantProduct, errPoolType)

    const escrow = this.escrowOf(pool)
    const a = this.readDeposit(depositA, escrow)
    const b = this.readDeposit(depositB, escrow)
    assert(a.asset === pool.assetA && b.asset === pool.assetB, errAssetNotInPool)

    const ctx = this.context(poolId, pool, Txn.sender, Txn.sender, pool.assetA, pool.assetB, a.amount, b.amount, 0, data)
    this.runHooks(poolId, pool, HookPointBeforeAddLiquidity, ctx, true)

    let usedA = a.amount
    let usedB = b.amount
    let minted: uint64 = 0

    if (pool.lpSupply === 0) {
      const liquidity = this.sqrtProduct(a.amount, b.amount)
      assert(liquidity > MinimumLiquidity, errInsufficientLiquidity)
      minted = liquidity - MinimumLiquidity
      pool.lpSupply = liquidity
    } else {
      assert(pool.reserveA > 0 && pool.reserveB > 0, errInsufficientLiquidity)
      const fromA = this.mulDiv(a.amount, pool.lpSupply, pool.reserveA)
      const fromB = this.mulDiv(b.amount, pool.lpSupply, pool.reserveB)
      if (fromA <= fromB) {
        minted = fromA
        const needB = this.mulDivUp(a.amount, pool.reserveB, pool.reserveA)
        usedB = needB < b.amount ? needB : b.amount
      } else {
        minted = fromB
        const needA = this.mulDivUp(b.amount, pool.reserveA, pool.reserveB)
        usedA = needA < a.amount ? needA : a.amount
      }
      pool.lpSupply += minted
    }

    assert(minted > 0 && minted >= minLpOut, errSlippage)
    pool.reserveA += usedA
    pool.reserveB += usedB

    ctx.amountIn = usedA
    ctx.amountOut = usedB
    ctx.lpAmount = minted
    ctx.lpSupply = pool.lpSupply
    this.runHooks(poolId, pool, HookPointAfterAddLiquidity, ctx, true)

    this.pools(poolId).value = clone(pool)

    itxn.assetTransfer({ assetReceiver: Txn.sender, xferAsset: pool.lpAsset, assetAmount: minted }).submit()
    if (a.amount > usedA) this.send(escrow, Txn.sender, pool.assetA, a.amount - usedA)
    if (b.amount > usedB) this.send(escrow, Txn.sender, pool.assetB, b.amount - usedB)

    emit<LiquidityEvent>({ poolId, caller: Txn.sender, amountA: usedA, amountB: usedB, lpAmount: minted, reserveA: pool.reserveA, reserveB: pool.reserveB })
    return minted
  }

  /** Bins: deposits go to the pool escrow, bin accounting lives in the bins book */
  addBinLiquidity(
    mbr: gtxn.PaymentTxn,
    depositA: gtxn.Transaction,
    depositB: gtxn.Transaction,
    poolId: uint64,
    lowerId: uint64,
    weightsA: uint64[],
    weightsB: uint64[],
    maxActiveDrift: uint64,
    expectedActiveId: uint64,
    data: bytes,
  ): uint64 {
    const pool = this.loadPool(poolId)
    assert(!pool.paused, errPoolPaused)
    assert(pool.poolType === PoolTypeBins, errPoolType)
    assert(mbr.receiver === Global.currentApplicationAddress, errInvalidPayment)
    const drift: uint64 = pool.activeId > expectedActiveId ? pool.activeId - expectedActiveId : expectedActiveId - pool.activeId
    assert(drift <= maxActiveDrift, errSlippage)

    const escrow = this.escrowOf(pool)
    const a = this.readDeposit(depositA, escrow)
    const b = this.readDeposit(depositB, escrow)
    assert(a.asset === pool.assetA && b.asset === pool.assetB, errAssetNotInPool)

    const ctx = this.context(poolId, pool, Txn.sender, Txn.sender, pool.assetA, pool.assetB, a.amount, b.amount, 0, data)
    this.runHooks(poolId, pool, HookPointBeforeAddLiquidity, ctx, true)

    const r = abiCall<typeof PluginDexBins.prototype.addLiquidity>({
      appId: this.bins.value,
      args: [
        itxn.payment({ receiver: this.bins.value.address, amount: mbr.amount }),
        poolId, pool.binStep, pool.activeId, Txn.sender, lowerId, a.amount, b.amount, weightsA, weightsB,
      ],
    }).returnValue

    pool.reserveA += r.amountA
    pool.reserveB += r.amountB
    ctx.amountIn = r.amountA
    ctx.amountOut = r.amountB
    this.runHooks(poolId, pool, HookPointAfterAddLiquidity, ctx, true)
    this.pools(poolId).value = clone(pool)

    if (a.amount > r.amountA) this.send(escrow, Txn.sender, pool.assetA, a.amount - r.amountA)
    if (b.amount > r.amountB) this.send(escrow, Txn.sender, pool.assetB, b.amount - r.amountB)
    emit<LiquidityEvent>({ poolId, caller: Txn.sender, amountA: r.amountA, amountB: r.amountB, lpAmount: 0, reserveA: pool.reserveA, reserveB: pool.reserveB })
    return r.amountA
  }

  removeBinLiquidity(poolId: uint64, lowerId: uint64, bps: uint64, minA: uint64, minB: uint64, data: bytes): RemoveResult {
    const pool = this.loadPool(poolId)
    assert(pool.poolType === PoolTypeBins && bps > 0 && bps <= BpsDivisor, errPoolType)
    const ctx = this.context(poolId, pool, Txn.sender, Txn.sender, pool.assetA, pool.assetB, 0, 0, bps, data)
    if (!pool.paused) this.runHooks(poolId, pool, HookPointBeforeRemoveLiquidity, ctx, true)

    const r = abiCall<typeof PluginDexBins.prototype.removeLiquidity>({
      appId: this.bins.value,
      args: [poolId, Txn.sender, lowerId, bps],
    }).returnValue
    assert(r.amountA >= minA && r.amountB >= minB, errSlippage)

    pool.reserveA -= r.amountA
    pool.reserveB -= r.amountB
    if (!pool.paused) {
      ctx.amountIn = r.amountA
      ctx.amountOut = r.amountB
      this.runHooks(poolId, pool, HookPointAfterRemoveLiquidity, ctx, true)
    }
    this.pools(poolId).value = clone(pool)

    const escrow = this.escrowOf(pool)
    if (r.amountA > 0) this.send(escrow, Txn.sender, pool.assetA, r.amountA)
    if (r.amountB > 0) this.send(escrow, Txn.sender, pool.assetB, r.amountB)
    emit<LiquidityEvent>({ poolId, caller: Txn.sender, amountA: r.amountA, amountB: r.amountB, lpAmount: 0, reserveA: pool.reserveA, reserveB: pool.reserveB })
    return { amountA: r.amountA, amountB: r.amountB }
  }

  removeLiquidity(lpTransfer: gtxn.AssetTransferTxn, poolId: uint64, minAmountA: uint64, minAmountB: uint64, data: bytes): RemoveResult {
    return this.removeCore(lpTransfer, poolId, minAmountA, minAmountB, data, true)
  }

  emergencyRemoveLiquidity(lpTransfer: gtxn.AssetTransferTxn, poolId: uint64, minAmountA: uint64, minAmountB: uint64): RemoveResult {
    assert(this.pools(poolId).exists, errPoolNotFound)
    assert(this.pools(poolId).value.paused, errPoolNotPaused)
    return this.removeCore(lpTransfer, poolId, minAmountA, minAmountB, Bytes(''), false)
  }

  swap(input: gtxn.Transaction, poolId: uint64, minAmountOut: uint64, recipient: Account, trader: Account, data: bytes): uint64 {
    const escrow = this.escrowOf(this.loadPool(poolId))
    const deposit = this.readDeposit(input, escrow)
    const to = this.resolveRecipient(recipient)
    const result = this.swapCore(poolId, deposit.asset, deposit.amount, this.resolveTrader(trader), to, data)
    assert(result.amountOut >= minAmountOut, errSlippage)
    this.send(escrow, to, result.assetOut, result.amountOut)
    return result.amountOut
  }

  swapRoute(input: gtxn.Transaction, route: uint64[], minAmountOut: uint64, recipient: Account, trader: Account, data: bytes): uint64 {
    assert(route.length > 0 && route.length <= MaxRouteHops, errInvalidRoute)
    let escrow = this.escrowOf(this.loadPool(route[0]))
    const deposit = this.readDeposit(input, escrow)
    const to = this.resolveRecipient(recipient)
    const who = this.resolveTrader(trader)
    let asset = deposit.asset
    let amount = deposit.amount
    for (let i: uint64 = 0; i < route.length; i += 1) {
      if (i > 0) ensureBudget(HopBudget, OpUpFeeSource.GroupCredit)
      const result = this.swapCore(route[i], asset, amount, who, to, data)
      asset = result.assetOut
      amount = result.amountOut
      if (i + 1 < route.length) {
        const next = this.escrowOf(this.loadPool(route[i + 1]))
        this.send(escrow, next, asset, amount)
        escrow = next
      }
    }
    assert(amount >= minAmountOut, errSlippage)
    this.send(escrow, to, asset, amount)
    return amount
  }

  @abimethod({ readonly: true })
  quote(poolId: uint64, assetIn: uint64, amountIn: uint64, trader: Account, data: bytes): uint64 {
    return this.quoteCore(poolId, assetIn, amountIn, trader, data).amountOut
  }

  @abimethod({ readonly: true })
  quoteRoute(route: uint64[], assetIn: uint64, amountIn: uint64, trader: Account, data: bytes): uint64 {
    assert(route.length > 0 && route.length <= MaxRouteHops, errInvalidRoute)
    let asset = assetIn
    let amount = amountIn
    for (let i: uint64 = 0; i < route.length; i += 1) {
      if (i > 0) ensureBudget(HopBudget, OpUpFeeSource.GroupCredit)
      const result = this.quoteCore(route[i], asset, amount, trader, data)
      asset = result.assetOut
      amount = result.amountOut
    }
    return amount
  }

  @abimethod({ readonly: true })
  getPool(poolId: uint64): PoolInfo {
    return this.loadPool(poolId)
  }
