import {
  Account,
  Application,
  assert,
  BoxMap,
  Bytes,
  clone,
  Contract,
  Global,
  GlobalState,
  gtxn,
  itxn,
  op,
  Txn,
  uint64,
} from '@algorandfoundation/algorand-typescript'
import { abiCall, abimethod, compileArc4 } from '@algorandfoundation/algorand-typescript/arc4'
import {
  BpsDivisor, HookPermAll, HookPermCustody, HookPermDynamicFee, HookPermExternal, HookPermHookFee, HookPointAll, HookPointBeforeSwap,
  HookPointQuote, MaxExternalPositions, MaxFeeBps, MaxHookFeeBps, MaxHooksPerPool, MaxPoolsPerPair, PoolTypeBins, PoolTypeLaunch,
} from './constants'
import {
  errAdminOnly, errExternalCap, errHookNotApproved, errHookPermission, errInvalidFee, errInvalidHookConfig, errInvalidPayment,
  errInvalidPool, errPoolType, errTooManyHooks, errTooManyPools,
} from './errors'
import { ExternalPosition, HookConfig, PairKey } from './types'
import { Escrow } from '../escrow/contract.algo'
import type { PluginDexV2 } from './core.algo'
import type { PluginDexBins } from './bins.algo'

/**
 * Pool factory + registry: hook / valuer approvals, pool creation (escrow create,
 * fund, opt in, rekey to core) and the pair index routers enumerate.
 */
export class PluginDexFactory extends Contract {
  admin = GlobalState<Account>({ key: 'admin' })
  core = GlobalState<Application>({ key: 'core' })
  bins = GlobalState<Application>({ key: 'bins' })
  approvedHooks = BoxMap<uint64, uint64>({ keyPrefix: 'a' })
  approvedValuers = BoxMap<uint64, uint64>({ keyPrefix: 'v' })
  pairPools = BoxMap<PairKey, uint64[]>({ keyPrefix: 'x' })

  @abimethod({ onCreate: 'require' })
  create(admin: Account): void {
    this.admin.value = admin
  }

  /** one-shot wiring after core and bins exist */
  init(core: Application, bins: Application): void {
    assert(Txn.sender === this.admin.value && this.core.value.id === 0, errAdminOnly)
    this.core.value = core
    this.bins.value = bins
  }

  private requireAdmin(): void {
    assert(Txn.sender === this.admin.value, errAdminOnly)
  }

  private validateHooks(hooks: HookConfig[]): uint64 {
    assert(hooks.length <= MaxHooksPerPool, errTooManyHooks)
    let points: uint64 = 0
    for (let i: uint64 = 0; i < hooks.length; i += 1) {
      const hook = clone(hooks[i])
      assert(hook.points !== 0 && (hook.points | HookPointAll) === HookPointAll, errInvalidHookConfig)
      assert(this.approvedHooks(hook.app).exists, errHookNotApproved)
      const approved = this.approvedHooks(hook.app).value
      assert((hook.perms | approved) === approved, errHookNotApproved)
      const pricing: uint64 = HookPermDynamicFee | HookPermHookFee
      if ((hook.perms & pricing) !== 0 && (hook.points & HookPointBeforeSwap) !== 0) {
        assert((hook.points & HookPointQuote) !== 0, errInvalidHookConfig)
      }
      for (let j: uint64 = 0; j < i; j += 1) {
        assert(hooks[j].app !== hook.app, errInvalidHookConfig)
      }
      points = points | hook.points
    }
    return points
  }

  setHookApproval(hook: uint64, perms: uint64): void {
    this.requireAdmin()
    if (perms === 0) {
      if (this.approvedHooks(hook).exists) this.approvedHooks(hook).delete()
      return
    }
    this.approvedHooks(hook).value = perms
  }

  setValuerApproval(valuer: uint64, approved: boolean): void {
    this.requireAdmin()
    if (!approved) {
      if (this.approvedValuers(valuer).exists) this.approvedValuers(valuer).delete()
      return
    }
    this.approvedValuers(valuer).value = 1
  }

  createPool(
    payment: gtxn.PaymentTxn,
    poolType: uint64,
    assetA: uint64,
    assetB: uint64,
    feeBps: uint64,
    maxHookFeeBps: uint64,
    curveParams: uint64[],
    hooks: HookConfig[],
    externals: ExternalPosition[],
    coreFunding: uint64,
  ): uint64 {
    assert(payment.receiver === Global.currentApplicationAddress, errInvalidPayment)
    assert(assetA < assetB, errInvalidPool)
    assert(feeBps <= MaxFeeBps && maxHookFeeBps <= MaxHookFeeBps, errInvalidFee)
    assert(poolType <= PoolTypeLaunch, errPoolType)
    const hookPoints = this.validateHooks(hooks)
    const mbrBefore = Global.currentApplicationAddress.minBalance

    assert(externals.length <= MaxExternalPositions && (externals.length === 0 || poolType !== PoolTypeBins), errExternalCap)
    for (let i: uint64 = 0; i < externals.length; i += 1) {
      const e = clone(externals[i])
      assert(this.approvedValuers(e.valuer).exists, errHookNotApproved)
      assert(e.booked === 0 && e.rate === 0 && e.capBps <= BpsDivisor, errExternalCap)
      let managed = false
      for (let j: uint64 = 0; j < hooks.length; j += 1) {
        const both: uint64 = HookPermCustody | HookPermExternal
        if (hooks[j].app === e.hook && (hooks[j].perms & both) === both) managed = true
      }
      assert(managed, errHookPermission)
    }

    let binPrice = op.bzero(16).toFixed({ length: 16 })
    if (poolType === PoolTypeBins) {
      binPrice = abiCall<typeof PluginDexBins.prototype.price>({ appId: this.bins.value, args: [curveParams[0], curveParams[1]] }).returnValue
    }

    const core = this.core.value
    const escrowContract = compileArc4(Escrow)
    const escrowApp = escrowContract.call.create({ args: [Bytes(op.itob(core.id))] }).itxn.createdApp
    const escrow = escrowApp.address
    const optIns: uint64 = assetA === 0 ? 1 : 2
    const escrowFunding: uint64 = Global.minBalance + optIns * Global.assetOptInMinBalance
    itxn.payment({ receiver: escrow, amount: escrowFunding }).submit()
    if (assetA !== 0) itxn.assetTransfer({ sender: escrow, assetReceiver: escrow, xferAsset: assetA, assetAmount: 0 }).submit()
    itxn.assetTransfer({ sender: escrow, assetReceiver: escrow, xferAsset: assetB, assetAmount: 0 }).submit()
    escrowContract.call.rekey({ appId: escrowApp, args: [core.address] })

    const poolId = abiCall<typeof PluginDexV2.prototype.registerPool>({
      appId: core,
      args: [
        itxn.payment({ receiver: core.address, amount: coreFunding }),
        poolType, assetA, assetB, escrowApp, feeBps, maxHookFeeBps, hookPoints, curveParams, Txn.sender, hooks, externals, binPrice,
      ],
    }).returnValue

    const pairKey: PairKey = { assetA, assetB }
    if (this.pairPools(pairKey).exists) {
      const ids = clone(this.pairPools(pairKey).value)
      assert(ids.length < MaxPoolsPerPair, errTooManyPools)
      ids.push(poolId)
      this.pairPools(pairKey).value = clone(ids)
    } else {
      this.pairPools(pairKey).value = [poolId]
    }

    // core refunds its unused funding to us (payment.sender of the forwarded txn)
    const required: uint64 = Global.currentApplicationAddress.minBalance - mbrBefore + escrowFunding + coreFunding
    assert(payment.amount >= required, errInvalidPayment)
    return poolId
  }

  @abimethod({ readonly: true })
  getPairPools(assetA: uint64, assetB: uint64): uint64[] {
    const key: PairKey = { assetA, assetB }
    return this.pairPools(key).exists ? clone(this.pairPools(key).value) : []
  }

  @abimethod({ readonly: true })
  getHookApproval(hook: uint64): uint64 {
    return this.approvedHooks(hook).exists ? this.approvedHooks(hook).value : 0
  }
}
