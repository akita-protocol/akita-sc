  @abimethod({ onCreate: 'require' })
  create(version: string, admin: Account): void {
    this.version.value = version
    this.admin.value = admin
    this.poolCursor.value = 0
    this.protocolFeeBps.value = 0
    this.treasury.value = admin
  }

  opUp(): void {}

  private requireAdmin(): void {
    loggedAssert(Txn.sender === this.admin.value, errAdminOnly)
  }

  private loadPool(poolId: uint64): PoolInfo {
    loggedAssert(this.pools(poolId).exists, errPoolNotFound)
    return clone(this.pools(poolId).value)
  }

  private escrowOf(pool: PoolInfo): Account {
    return Application(pool.escrow).address
  }

  private holding(account: Account, asset: uint64): uint64 {
    if (asset === 0) {
      return account.balance - account.minBalance
    }
    const [balance, optedIn] = op.AssetHolding.assetBalance(account, asset)
    return optedIn ? balance : 0
  }

  private send(sender: Account, receiver: Account, asset: uint64, amount: uint64): void {
    if (asset === 0) {
      itxn.payment({ sender, receiver, amount }).submit()
    } else {
      itxn.assetTransfer({ sender, assetReceiver: receiver, xferAsset: asset, assetAmount: amount }).submit()
    }
  }

  private readDeposit(txn: gtxn.Transaction, receiver: Account): Deposit {
    loggedAssert(txn.rekeyTo === Global.zeroAddress, errInvalidInput)
    if (txn.type === TransactionType.Payment) {
      loggedAssert(txn.receiver === receiver, errInvalidInput)
      loggedAssert(txn.closeRemainderTo === Global.zeroAddress, errInvalidInput)
      loggedAssert(txn.amount > 0, errInvalidInput)
      return { asset: 0, amount: txn.amount }
    }
    loggedAssert(txn.type === TransactionType.AssetTransfer, errInvalidInput)
    loggedAssert(txn.assetReceiver === receiver, errInvalidInput)
    loggedAssert(txn.assetCloseTo === Global.zeroAddress, errInvalidInput)
    loggedAssert(txn.assetSender === Global.zeroAddress, errInvalidInput)
    loggedAssert(txn.assetAmount > 0, errInvalidInput)
    return { asset: txn.xferAsset.id, amount: txn.assetAmount }
  }

  // MATH ----------------------------------------------------------------------------------------

  private mulDiv(a: uint64, b: uint64, c: uint64): uint64 {
    return op.divw(...op.mulw(a, b), c)
  }

  private mulDivUp(a: uint64, b: uint64, c: uint64): uint64 {
    const [qHi, qLo, rHi, rLo] = op.divmodw(...op.mulw(a, b), 0, c)
    loggedAssert(qHi === 0, errInvalidInput)
    return rHi === 0 && rLo === 0 ? qLo : qLo + 1
  }

  private sqrtProduct(a: uint64, b: uint64): uint64 {
    return op.btoi(Bytes(op.bsqrt(BigUint(a) * BigUint(b))))
  }

  /** Q64.64 price of A in B, padded to 16 bytes */
  private cpPrice(reserveA: uint64, reserveB: uint64): bytes<16> {
    if (reserveA === 0) {
      return op.bzero(16).toFixed({ length: 16 })
    }
    const p: biguint = (BigUint(reserveB) * BigUint(Bytes.fromHex('010000000000000000'))) / BigUint(reserveA)
    return op.bzero(16).bitwiseOr(Bytes(p)).toFixed({ length: 16 })
  }

  private applyDelta(reserve: uint64, before: uint64, after: uint64): uint64 {
    if (after >= before) {
      return reserve + (after - before)
    }
    const loss: uint64 = before - after
    loggedAssert(reserve >= loss, errCustodyInvariant)
    return reserve - loss
  }

  private assertKNotDecreased(oldA: uint64, oldB: uint64, newA: uint64, newB: uint64): void {
    if (oldA === 0 || oldB === 0) {
      loggedAssert(newA >= oldA && newB >= oldB, errCustodyInvariant)
      return
    }
    loggedAssert(BigUint(newA) * BigUint(newB) >= BigUint(oldA) * BigUint(oldB), errCustodyInvariant)
  }

  // CURVE ---------------------------------------------------------------------------------------

  private callCurve(pool: PoolInfo, req: CurveRequest): CurveResult {
    return abiCall<typeof PluginDexCurve.prototype.akitaDexCurve>({
      appId: pool.curve,
      args: [Global.currentApplicationId, req],
    }).returnValue
  }

  private curveRequest(op_: uint64, poolId: uint64, pool: PoolInfo): CurveRequest {
    return {
      op: op_,
      poolId,
      caller: Txn.sender,
      inIsA: true,
      amountA: 0,
      amountB: 0,
      negA: false,
      negB: false,
      lpAmount: 0,
      lpSupply: pool.lpSupply,
      feeBps: pool.feeBps,
      protocolFeeBps: this.protocolFeeBps.value,
      reserveA: pool.reserveA,
      reserveB: pool.reserveB,
      args: Bytes(''),
    }
  }

  /** the curve's aggregate must always equal the DEX ledger, and price/state are cached for hooks */
  private bookCurve(pool: PoolInfo, res: CurveResult): void {
    loggedAssert(res.reserveA === pool.reserveA && res.reserveB === pool.reserveB, errCurve)
    pool.priceX64 = res.priceX64
    pool.activeId = res.activeId
    pool.lastUpdate = Global.latestTimestamp
    if ((res.flags & CurveFlagClose) !== 0) pool.closed = true
  }

  // HOOKS ---------------------------------------------------------------------------------------

  private approval(app: uint64): HookApproval {
    loggedAssert(this.approvedHooks(app).exists, errHookNotApproved)
    return clone(this.approvedHooks(app).value)
  }

  private validateHooks(hooks: HookConfig[]): uint64 {
    loggedAssert(hooks.length <= MaxHooksPerPool, errTooManyHooks)
    let points: uint64 = 0
    for (let i: uint64 = 0; i < hooks.length; i += 1) {
      const hook = clone(hooks[i])
      loggedAssert(hook.points !== 0 && (hook.points | HookPointAll) === HookPointAll, errInvalidHookConfig)
      // the curve permission is pool-level, never part of the stack
      loggedAssert((hook.perms | HookPermAll) === HookPermAll && (hook.perms & HookPermCurve) === 0, errInvalidHookConfig)
      const approved = this.approval(hook.app).perms
      loggedAssert((hook.perms | approved) === approved, errHookNotApproved)
      if ((hook.perms & HookPermExternal) !== 0) {
        loggedAssert((hook.perms & HookPermCustody) !== 0, errInvalidHookConfig)
      }
      const pricing: uint64 = HookPermDynamicFee | HookPermHookFee
      if ((hook.perms & pricing) !== 0 && (hook.points & HookPointBeforeSwap) !== 0) {
        loggedAssert((hook.points & HookPointQuote) !== 0, errInvalidHookConfig)
      }
      for (let j: uint64 = 0; j < i; j += 1) {
        loggedAssert(hooks[j].app !== hook.app, errInvalidHookConfig)
      }
      points = points | hook.points
    }
    return points
  }

  /**
   * Book a custody hook's balance changes. Principal it declares as moved to/from an
   * external position does not count as a reserve change; everything else does and must
   * pass the pool type's guard (k for native CP, a curve Settle for curve pools).
   */
  private settleCustody(
    poolId: uint64,
    pool: PoolInfo,
    run: HookRun,
    hook: HookConfig,
    extOutA: uint64,
    extOutB: uint64,
    extInA: uint64,
    extInB: uint64,
    balA: uint64,
    balB: uint64,
  ): void {
    const escrow = this.escrowOf(pool)
    const afterA: uint64 = this.holding(escrow, pool.assetA) + extOutA
    const afterB: uint64 = this.holding(escrow, pool.assetB) + extOutB
    const beforeA: uint64 = balA + extInA
    const beforeB: uint64 = balB + extInB

    if (extOutA + extOutB + extInA + extInB > 0) {
      loggedAssert((hook.perms & HookPermExternal) !== 0, errHookPermission)
      const key: ExtKey = { poolId, hook: hook.app }
      let pos: ExternalPosition = { principalA: 0, principalB: 0, receipt: 0 }
      if (this.externals(key).exists) pos = clone(this.externals(key).value)
      loggedAssert(pos.principalA + extOutA >= extInA && pos.principalB + extOutB >= extInB, errExternal)
      pos.principalA = pos.principalA + extOutA - extInA
      pos.principalB = pos.principalB + extOutB - extInB
      run.externalA = run.externalA + extOutA - extInA
      run.externalB = run.externalB + extOutB - extInB
      const cap = this.approval(hook.app).extCapBps
      loggedAssert(pos.principalA <= this.mulDiv(run.reserveA, cap, BpsDivisor), errExternal)
      loggedAssert(pos.principalB <= this.mulDiv(run.reserveB, cap, BpsDivisor), errExternal)
      this.externals(key).value = clone(pos)
    }

    const newA = this.applyDelta(run.reserveA, beforeA, afterA)
    const newB = this.applyDelta(run.reserveB, beforeB, afterB)
    if (pool.curve === 0) {
      this.assertKNotDecreased(run.reserveA, run.reserveB, newA, newB)
    } else if (newA !== run.reserveA || newB !== run.reserveB) {
      const req = this.curveRequest(CurveOpSettle, poolId, pool)
      req.reserveA = run.reserveA
      req.reserveB = run.reserveB
      req.negA = newA < run.reserveA
      req.negB = newB < run.reserveB
      req.amountA = req.negA ? run.reserveA - newA : newA - run.reserveA
      req.amountB = req.negB ? run.reserveB - newB : newB - run.reserveB
      loggedAssert(!(req.negA && req.negB), errCustodyInvariant)
      const res = this.callCurve(pool, req)
      pool.reserveA = newA
      pool.reserveB = newB
      this.bookCurve(pool, res)
    }
    run.reserveA = newA
    run.reserveB = newB
    emit<CustodyEvent>({ poolId, hook: hook.app, point: 0, reserveA: newA, reserveB: newB })
  }

  private runHooks(poolId: uint64, pool: PoolInfo, point: uint64, base: HookContext, inIsA: boolean): HookRun {
    const run: HookRun = {
      reserveA: pool.reserveA,
      reserveB: pool.reserveB,
      externalA: pool.externalA,
      externalB: pool.externalB,
      feeBps: base.feeBps,
      hookFees: 0,
    }
    if ((pool.hookPoints & point) === 0) {
      return run
    }

    const escrow = this.escrowOf(pool)
    const hooks = clone(this.poolHooks(poolId).value)
    const maxHookFees = this.mulDiv(base.amountIn, pool.maxHookFeeBps, BpsDivisor)

    for (let i: uint64 = 0; i < hooks.length; i += 1) {
      const hook = clone(hooks[i])
      if ((hook.points & point) === 0) {
        continue
      }
      const ctx = clone(base)
      ctx.point = point
      ctx.feeBps = run.feeBps
      ctx.reserveIn = inIsA ? run.reserveA : run.reserveB
      ctx.reserveOut = inIsA ? run.reserveB : run.reserveA
      ctx.priceAfterX64 = pool.priceX64
      ctx.activeIdAfter = pool.activeId

      const hookApp = Application(hook.app)
      const custody = (hook.perms & HookPermCustody) !== 0 && point !== HookPointQuote

      let balanceA: uint64 = 0
      let balanceB: uint64 = 0
      if (custody) {
        balanceA = this.holding(escrow, pool.assetA)
        balanceB = this.holding(escrow, pool.assetB)
        itxn.payment({ sender: escrow, receiver: escrow, amount: 0, rekeyTo: hookApp.address }).submit()
      }

      const result = abiCall<typeof PluginDexHook.prototype.akitaDexHook>({
        appId: hookApp,
        args: [Global.currentApplicationId, i, ctx],
      }).returnValue

      const ext = (result.flags & HookResultExternal) !== 0
      if (custody) {
        loggedAssert(escrow.authAddress === Global.currentApplicationAddress, errMissingRekeyBack)
        this.settleCustody(
          poolId,
          pool,
          run,
          hook,
          ext ? result.extOutA : 0,
          ext ? result.extOutB : 0,
          ext ? result.extInA : 0,
          ext ? result.extInB : 0,
          balanceA,
          balanceB,
        )
      } else {
        loggedAssert(!ext, errHookPermission)
      }

      const pricingPoint = point === HookPointBeforeSwap || point === HookPointQuote
      if ((result.flags & HookResultSetFee) !== 0) {
        loggedAssert(pricingPoint && (hook.perms & HookPermDynamicFee) !== 0, errHookPermission)
        loggedAssert(result.feeBps <= MaxFeeBps, errInvalidFee)
        run.feeBps = result.feeBps
      }
      if (result.hookFee > 0) {
        loggedAssert(pricingPoint && (hook.perms & HookPermHookFee) !== 0, errHookPermission)
        run.hookFees += result.hookFee
        loggedAssert(run.hookFees <= maxHookFees, errHookFeeTooHigh)
        if (point === HookPointBeforeSwap) {
          this.send(escrow, hookApp.address, base.assetIn, result.hookFee)
        }
      }
    }
    return run
  }

  private applyRun(pool: PoolInfo, run: HookRun): void {
    pool.reserveA = run.reserveA
    pool.reserveB = run.reserveB
    pool.externalA = run.externalA
    pool.externalB = run.externalB
  }

  private context(
    poolId: uint64,
    pool: PoolInfo,
    recipient: Account,
    assetIn: uint64,
    assetOut: uint64,
    amountIn: uint64,
    amountOut: uint64,
    lpAmount: uint64,
    data: bytes,
  ): HookContext {
    return {
      point: 0,
      poolId,
      escrow: this.escrowOf(pool),
      caller: Txn.sender,
      recipient,
      assetIn,
      assetOut,
      amountIn,
      amountOut,
      reserveIn: 0,
      reserveOut: 0,
      lpAmount,
      lpSupply: pool.lpSupply,
      feeBps: pool.feeBps,
      data,
      curve: pool.curve,
      priceX64: pool.priceX64,
      priceAfterX64: pool.priceX64,
      activeId: pool.activeId,
      activeIdAfter: pool.activeId,
      lastUpdate: pool.lastUpdate,
    }
  }

  // SWAPS ---------------------------------------------------------------------------------------

  /**
   * Price `amountIn` (net of hook fees) against the pool type and book it. Native CP is
   * computed here; a curve pool's app returns the amounts and the DEX checks them:
   * full fill, solvency against in-escrow reserves, protocol fee, price band, and that
   * the curve's aggregate equals the ledger.
   */
  private price(poolId: uint64, pool: PoolInfo, inIsA: boolean, amountIn: uint64, feeBps: uint64, commit: boolean): uint64 {
    loggedAssert(amountIn > 0, errInvalidInput)
    const reserveIn: uint64 = inIsA ? pool.reserveA : pool.reserveB
    const reserveOut: uint64 = inIsA ? pool.reserveB : pool.reserveA
    const liquidOut: uint64 = reserveOut - (inIsA ? pool.externalB : pool.externalA)

    let amountOut: uint64 = 0
    let lpFee: uint64 = 0
    let res: CurveResult = { amountA: 0, amountB: 0, lpFee: 0, shares: 0, priceX64: pool.priceX64, activeId: 0, reserveA: 0, reserveB: 0, flags: 0 }
    if (pool.curve === 0) {
      loggedAssert(reserveIn > 0 && reserveOut > 0, errInsufficientLiquidity)
      lpFee = this.mulDiv(amountIn, feeBps, BpsDivisor)
      const effectiveIn: uint64 = amountIn - lpFee
      amountOut = this.mulDiv(effectiveIn, reserveOut, reserveIn + effectiveIn)
    } else {
      const req = this.curveRequest(commit ? CurveOpSwap : CurveOpQuote, poolId, pool)
      req.inIsA = inIsA
      req.amountA = inIsA ? amountIn : 0
      req.amountB = inIsA ? 0 : amountIn
      req.feeBps = feeBps
      const r = this.callCurve(pool, req)
      loggedAssert((inIsA ? r.amountA : r.amountB) === amountIn, errCurve)
      amountOut = inIsA ? r.amountB : r.amountA
      lpFee = r.lpFee
      loggedAssert(lpFee <= amountIn, errCurve)
      // price band: no fill better than the pre-trade marginal price, and price moves against the taker
      const one = BigUint(Bytes.fromHex('010000000000000000'))
      const pre = BigUint(pool.priceX64)
      const eff = BigUint(amountIn - lpFee)
      if (inIsA) {
        loggedAssert(BigUint(amountOut) * one <= eff * pre && BigUint(r.priceX64) <= pre, errCurve)
      } else {
        loggedAssert(BigUint(amountOut) * pre <= eff * one && BigUint(r.priceX64) >= pre, errCurve)
      }
      res = clone(r)
    }
    loggedAssert(amountOut > 0 && amountOut < liquidOut, errInsufficientLiquidity)

    if (commit) {
      const protocolFee = this.mulDiv(lpFee, this.protocolFeeBps.value, BpsDivisor)
      if (inIsA) {
        pool.reserveA += amountIn - protocolFee
        pool.reserveB -= amountOut
        pool.protocolFeesA += protocolFee
      } else {
        pool.reserveB += amountIn - protocolFee
        pool.reserveA -= amountOut
        pool.protocolFeesB += protocolFee
      }
      if (pool.curve !== 0) {
        this.bookCurve(pool, res)
      } else {
        pool.priceX64 = this.cpPrice(pool.reserveA, pool.reserveB)
        pool.lastUpdate = Global.latestTimestamp
      }
    }
    return amountOut
  }

  private swapCore(poolId: uint64, assetIn: uint64, amountIn: uint64, recipient: Account, data: bytes): SwapResult {
    const pool = this.loadPool(poolId)
    loggedAssert(!pool.paused && !pool.closed, errPoolPaused)
    const inIsA = assetIn === pool.assetA
    loggedAssert(inIsA || assetIn === pool.assetB, errAssetNotInPool)
    const assetOut = inIsA ? pool.assetB : pool.assetA

    const ctx = this.context(poolId, pool, recipient, assetIn, assetOut, amountIn, 0, 0, data)
    const before = this.runHooks(poolId, pool, HookPointBeforeSwap, ctx, inIsA)
    this.applyRun(pool, before)
    // a custody hook may have moved the price; hooks after it see the new one
    ctx.priceX64 = pool.priceX64
    ctx.activeId = pool.activeId

    const amountOut = this.price(poolId, pool, inIsA, amountIn - before.hookFees, before.feeBps, true)

    ctx.amountOut = amountOut
    ctx.feeBps = before.feeBps
    const after = this.runHooks(poolId, pool, HookPointAfterSwap, ctx, inIsA)
    this.applyRun(pool, after)

    this.pools(poolId).value = clone(pool)
    emit<SwapEvent>({
      poolId,
      caller: Txn.sender,
      assetIn,
      amountIn,
      assetOut,
      amountOut,
      feeBps: before.feeBps,
      hookFees: before.hookFees,
      reserveA: pool.reserveA,
      reserveB: pool.reserveB,
    })
    return { assetOut, amountOut }
  }

  private quoteCore(poolId: uint64, assetIn: uint64, amountIn: uint64): SwapResult {
    const pool = this.loadPool(poolId)
    loggedAssert(!pool.paused && !pool.closed, errPoolPaused)
    const inIsA = assetIn === pool.assetA
    loggedAssert(inIsA || assetIn === pool.assetB, errAssetNotInPool)
    const assetOut = inIsA ? pool.assetB : pool.assetA
    const ctx = this.context(poolId, pool, Txn.sender, assetIn, assetOut, amountIn, 0, 0, Bytes(''))
    const run = this.runHooks(poolId, pool, HookPointQuote, ctx, inIsA)
    return { assetOut, amountOut: this.price(poolId, pool, inIsA, amountIn - run.hookFees, run.feeBps, false) }
  }

  private resolveRecipient(recipient: Account): Account {
    return recipient === Global.zeroAddress ? Txn.sender : recipient
  }

  // LIQUIDITY -----------------------------------------------------------------------------------

  /** shared add path: fungible LP for native CP and fungible curve pools, positions for curve-managed */
  private addCore(
    depositA: gtxn.Transaction,
    depositB: gtxn.Transaction,
    poolId: uint64,
    minOut: uint64,
    args: bytes,
    data: bytes,
  ): uint64 {
    const pool = this.loadPool(poolId)
    loggedAssert(!pool.paused && !pool.closed, errPoolPaused)
    const escrow = this.escrowOf(pool)
    const a = this.readDeposit(depositA, escrow)
    const b = this.readDeposit(depositB, escrow)
    loggedAssert(a.asset === pool.assetA && b.asset === pool.assetB, errAssetNotInPool)

    const ctx = this.context(poolId, pool, Txn.sender, pool.assetA, pool.assetB, a.amount, b.amount, 0, data)
    this.applyRun(pool, this.runHooks(poolId, pool, HookPointBeforeAddLiquidity, ctx, true))

    let usedA = a.amount
    let usedB = b.amount
    let minted: uint64 = 0

    if (pool.curve !== 0) {
      const req = this.curveRequest(CurveOpAdd, poolId, pool)
      req.amountA = a.amount
      req.amountB = b.amount
      req.args = args
      const r = this.callCurve(pool, req)
      loggedAssert(r.amountA <= a.amount && r.amountB <= b.amount, errCurve)
      usedA = r.amountA
      usedB = r.amountB
      minted = r.shares
      pool.reserveA += usedA
      pool.reserveB += usedB
      if (pool.lpMode === LpModeFungible) pool.lpSupply += minted
      this.bookCurve(pool, r)
    } else {
      if (pool.lpSupply === 0) {
        const liquidity = this.sqrtProduct(a.amount, b.amount)
        loggedAssert(liquidity > MinimumLiquidity, errInsufficientLiquidity)
        minted = liquidity - MinimumLiquidity
        pool.lpSupply = liquidity
      } else {
        loggedAssert(pool.reserveA > 0 && pool.reserveB > 0, errInsufficientLiquidity)
        const fromA = this.mulDiv(a.amount, pool.lpSupply, pool.reserveA)
        const fromB = this.mulDiv(b.amount, pool.lpSupply, pool.reserveB)
        if (fromA <= fromB) {
          minted = fromA
          const needB = this.mulDivUp(a.amount, pool.reserveB, pool.reserveA)
          usedB = needB < b.amount ? needB : b.amount
        } else {
          minted = fromB
          const needA = this.mulDivUp(b.amount, pool.reserveA, pool.reserveB)
          usedA = needA < a.amount ? needA : a.amount
        }
        pool.lpSupply += minted
      }
      pool.reserveA += usedA
      pool.reserveB += usedB
      pool.priceX64 = this.cpPrice(pool.reserveA, pool.reserveB)
    }
    loggedAssert(minted > 0 && minted >= minOut, errSlippage)

    ctx.amountIn = usedA
    ctx.amountOut = usedB
    ctx.lpAmount = minted
    ctx.lpSupply = pool.lpSupply
    this.applyRun(pool, this.runHooks(poolId, pool, HookPointAfterAddLiquidity, ctx, true))
    this.pools(poolId).value = clone(pool)

    if (pool.lpMode === LpModeFungible) {
      itxn.assetTransfer({ assetReceiver: Txn.sender, xferAsset: pool.lpAsset, assetAmount: minted }).submit()
    }
    if (a.amount > usedA) this.send(escrow, Txn.sender, pool.assetA, a.amount - usedA)
    if (b.amount > usedB) this.send(escrow, Txn.sender, pool.assetB, b.amount - usedB)
    emit<LiquidityEvent>({ poolId, caller: Txn.sender, amountA: usedA, amountB: usedB, lpAmount: minted, reserveA: pool.reserveA, reserveB: pool.reserveB })
    return minted
  }

  private removeCore(
    poolId: uint64,
    lpAmount: uint64,
    minAmountA: uint64,
    minAmountB: uint64,
    args: bytes,
    data: bytes,
    withHooks: boolean,
  ): RemoveResult {
    const pool = this.loadPool(poolId)
    const ctx = this.context(poolId, pool, Txn.sender, pool.assetA, pool.assetB, 0, 0, lpAmount, data)
    if (withHooks) {
      this.applyRun(pool, this.runHooks(poolId, pool, HookPointBeforeRemoveLiquidity, ctx, true))
    }

    let amountA: uint64 = 0
    let amountB: uint64 = 0
    if (pool.curve !== 0) {
      const req = this.curveRequest(CurveOpRemove, poolId, pool)
      req.lpAmount = lpAmount
      req.args = args
      const r = this.callCurve(pool, req)
      amountA = r.amountA
      amountB = r.amountB
      pool.reserveA -= amountA
      pool.reserveB -= amountB
      if (pool.lpMode === LpModeFungible) {
        loggedAssert(r.shares === lpAmount, errCurve)
        pool.lpSupply -= lpAmount
      }
      this.bookCurve(pool, r)
    } else {
      amountA = this.mulDiv(lpAmount, pool.reserveA, pool.lpSupply)
      amountB = this.mulDiv(lpAmount, pool.reserveB, pool.lpSupply)
      pool.reserveA -= amountA
      pool.reserveB -= amountB
      pool.lpSupply -= lpAmount
    }
    loggedAssert(amountA >= minAmountA && amountB >= minAmountB, errSlippage)
    // only in-escrow liquidity can leave; external principal must be recalled first (BeforeRemove)
    loggedAssert(pool.reserveA >= pool.externalA && pool.reserveB >= pool.externalB, errExternal)

    if (withHooks) {
      ctx.amountIn = amountA
      ctx.amountOut = amountB
      ctx.lpSupply = pool.lpSupply
      this.applyRun(pool, this.runHooks(poolId, pool, HookPointAfterRemoveLiquidity, ctx, true))
    }
    this.pools(poolId).value = clone(pool)

    const escrow = this.escrowOf(pool)
    if (amountA > 0) this.send(escrow, Txn.sender, pool.assetA, amountA)
    if (amountB > 0) this.send(escrow, Txn.sender, pool.assetB, amountB)
    emit<LiquidityEvent>({ poolId, caller: Txn.sender, amountA, amountB, lpAmount, reserveA: pool.reserveA, reserveB: pool.reserveB })
    return { amountA, amountB }
  }

  private readLpTransfer(lpTransfer: gtxn.AssetTransferTxn, poolId: uint64): uint64 {
    loggedAssert(lpTransfer.xferAsset.id === this.loadPool(poolId).lpAsset, errInvalidInput)
    loggedAssert(lpTransfer.assetReceiver === Global.currentApplicationAddress, errInvalidInput)
    loggedAssert(lpTransfer.assetCloseTo === Global.zeroAddress, errInvalidInput)
    loggedAssert(lpTransfer.assetSender === Global.zeroAddress, errInvalidInput)
    loggedAssert(lpTransfer.rekeyTo === Global.zeroAddress, errInvalidInput)
    loggedAssert(lpTransfer.assetAmount > 0, errInvalidInput)
    return lpTransfer.assetAmount
  }

  setAdmin(admin: Account): void {
    this.requireAdmin()
    this.admin.value = admin
  }

  setProtocolFee(protocolFeeBps: uint64, treasury: Account): void {
    this.requireAdmin()
    loggedAssert(protocolFeeBps <= MaxProtocolFeeBps, errInvalidFee)
    this.protocolFeeBps.value = protocolFeeBps
    this.treasury.value = treasury
  }

  setHookApproval(hook: uint64, perms: uint64, extCapBps: uint64): void {
    this.requireAdmin()
    loggedAssert((perms | HookPermAll) === HookPermAll && extCapBps <= BpsDivisor, errInvalidHookConfig)
    if (perms === 0) {
      if (this.approvedHooks(hook).exists) this.approvedHooks(hook).delete()
      return
    }
    this.approvedHooks(hook).value = { perms, extCapBps }
  }

  setPoolPaused(poolId: uint64, paused: boolean): void {
    this.requireAdmin()
    loggedAssert(this.pools(poolId).exists, errPoolNotFound)
    this.pools(poolId).value.paused = paused
  }

  collectProtocolFees(poolId: uint64): void {
    const pool = this.loadPool(poolId)
    const escrow = this.escrowOf(pool)
    const feesA = pool.protocolFeesA
    const feesB = pool.protocolFeesB
    this.pools(poolId).value.protocolFeesA = 0
    this.pools(poolId).value.protocolFeesB = 0
    if (feesA > 0) this.send(escrow, this.treasury.value, pool.assetA, feesA)
    if (feesB > 0) this.send(escrow, this.treasury.value, pool.assetB, feesB)
  }

  /**
   * Book unaccounted escrow balance (consensus rewards, direct donations) into reserves.
   * Native CP: plain donation. Curve pools: a Settle donation the curve distributes.
   */
  sync(poolId: uint64): void {
    const pool = this.loadPool(poolId)
    const escrow = this.escrowOf(pool)
    const expectA: uint64 = pool.reserveA - pool.externalA + pool.protocolFeesA
    const expectB: uint64 = pool.reserveB - pool.externalB + pool.protocolFeesB
    const hA = this.holding(escrow, pool.assetA)
    const hB = this.holding(escrow, pool.assetB)
    const addA: uint64 = hA > expectA ? hA - expectA : 0
    const addB: uint64 = hB > expectB ? hB - expectB : 0
    if (addA + addB === 0) return
    if (pool.curve !== 0) {
      const req = this.curveRequest(CurveOpSettle, poolId, pool)
      req.amountA = addA
      req.amountB = addB
      const res = this.callCurve(pool, req)
      pool.reserveA += addA
      pool.reserveB += addB
      this.bookCurve(pool, res)
    } else {
      pool.reserveA += addA
      pool.reserveB += addB
      pool.priceX64 = this.cpPrice(pool.reserveA, pool.reserveB)
    }
    this.pools(poolId).value = clone(pool)
  }

  createPool(
    payment: gtxn.PaymentTxn,
    assetA: uint64,
    assetB: uint64,
    feeBps: uint64,
    maxHookFeeBps: uint64,
    hooks: HookConfig[],
    curve: uint64,
    lpMode: uint64,
    curveArgs: bytes,
  ): uint64 {
    loggedAssert(payment.receiver === Global.currentApplicationAddress, errInvalidPayment)
    loggedAssert(assetA < assetB, errInvalidPool)
    loggedAssert(feeBps <= MaxFeeBps, errInvalidFee)
    loggedAssert(maxHookFeeBps <= MaxHookFeeBps, errInvalidFee)
    if (curve !== 0) {
      loggedAssert((this.approval(curve).perms & HookPermCurve) !== 0, errHookNotApproved)
      loggedAssert(lpMode <= LpModeCurve, errInvalidPool)
    } else {
      loggedAssert(lpMode === LpModeFungible, errInvalidPool)
    }

    const hookPoints = this.validateHooks(hooks)
    const dexMbrBefore = Global.currentApplicationAddress.minBalance
    const poolId = this.poolCursor.value
    this.poolCursor.value = poolId + 1

    const escrowContract = compileArc4(Escrow)
    const escrowApp = escrowContract.call.create({ args: [Bytes(op.itob(Global.currentApplicationId.id))] }).itxn.createdApp
    const escrow = escrowApp.address
    const optIns: uint64 = assetA === 0 ? 1 : 2
    const escrowFunding: uint64 = Global.minBalance + optIns * Global.assetOptInMinBalance
    itxn.payment({ receiver: escrow, amount: escrowFunding }).submit()
    escrowContract.call.rekey({ appId: escrowApp, args: [Global.currentApplicationAddress] })
    if (assetA !== 0) this.send(escrow, escrow, assetA, 0)
    this.send(escrow, escrow, assetB, 0)

    let lpAsset: uint64 = 0
    if (lpMode === LpModeFungible) {
      lpAsset = itxn
        .assetConfig({
          total: LPAssetTotal,
          decimals: LPAssetDecimals,
          unitName: LPAssetUnitName,
          assetName: LPAssetName,
          reserve: Global.currentApplicationAddress,
        })
        .submit().createdAsset.id
    }

    const pool: PoolInfo = {
      assetA,
      assetB,
      escrow: escrowApp.id,
      lpAsset,
      reserveA: 0,
      reserveB: 0,
      lpSupply: 0,
      feeBps,
      maxHookFeeBps,
      hookPoints,
      protocolFeesA: 0,
      protocolFeesB: 0,
      creator: Txn.sender,
      paused: false,
      curve,
      lpMode,
      closed: false,
      priceX64: op.bzero(16).toFixed({ length: 16 }),
      activeId: 0,
      lastUpdate: Global.latestTimestamp,
      externalA: 0,
      externalB: 0,
    }

    if (curve !== 0) {
      const req = this.curveRequest(CurveOpInit, poolId, pool)
      req.args = curveArgs
      this.bookCurve(pool, this.callCurve(pool, req))
    }
    this.pools(poolId).value = clone(pool)

    if (hooks.length > 0) {
      this.poolHooks(poolId).value = clone(hooks)
    }

    const pairKey: PairKey = { assetA, assetB }
    if (this.pairPools(pairKey).exists) {
      const ids = clone(this.pairPools(pairKey).value)
      loggedAssert(ids.length < MaxPoolsPerPair, errTooManyPools)
      ids.push(poolId)
      this.pairPools(pairKey).value = clone(ids)
    } else {
      this.pairPools(pairKey).value = [poolId]
    }

    const ctx = this.context(poolId, pool, Txn.sender, assetA, assetB, 0, 0, 0, Bytes(''))
    const init = this.runHooks(poolId, pool, HookPointAfterInitialize, ctx, true)
    if (init.reserveA !== 0 || init.reserveB !== 0) {
      this.pools(poolId).value.reserveA = init.reserveA
      this.pools(poolId).value.reserveB = init.reserveB
    }

    const required: uint64 = Global.currentApplicationAddress.minBalance - dexMbrBefore + escrowFunding
    loggedAssert(payment.amount >= required, errInvalidPayment)
    if (payment.amount > required) {
      itxn.payment({ receiver: payment.sender, amount: payment.amount - required }).submit()
    }
    return poolId
  }

  addLiquidity(depositA: gtxn.Transaction, depositB: gtxn.Transaction, poolId: uint64, minLpOut: uint64, data: bytes): uint64 {
    return this.addCore(depositA, depositB, poolId, minLpOut, Bytes(''), data)
  }

  /** curve-managed positions: `curveArgs` carries the position shape (e.g. bin distribution) */
  addLiquidityCurve(
    depositA: gtxn.Transaction,
    depositB: gtxn.Transaction,
    poolId: uint64,
    minShares: uint64,
    curveArgs: bytes,
    data: bytes,
  ): uint64 {
    return this.addCore(depositA, depositB, poolId, minShares, curveArgs, data)
  }

  removeLiquidity(lpTransfer: gtxn.AssetTransferTxn, poolId: uint64, minAmountA: uint64, minAmountB: uint64, data: bytes): RemoveResult {
    return this.removeCore(poolId, this.readLpTransfer(lpTransfer, poolId), minAmountA, minAmountB, Bytes(''), data, true)
  }

  removeLiquidityCurve(poolId: uint64, minAmountA: uint64, minAmountB: uint64, curveArgs: bytes, data: bytes): RemoveResult {
    loggedAssert(this.loadPool(poolId).lpMode === LpModeCurve, errInvalidPool)
    return this.removeCore(poolId, 0, minAmountA, minAmountB, curveArgs, data, true)
  }

  emergencyRemoveLiquidity(lpTransfer: gtxn.AssetTransferTxn, poolId: uint64, minAmountA: uint64, minAmountB: uint64): RemoveResult {
    loggedAssert(this.pools(poolId).exists, errPoolNotFound)
    loggedAssert(this.pools(poolId).value.paused, errPoolNotPaused)
    return this.removeCore(poolId, this.readLpTransfer(lpTransfer, poolId), minAmountA, minAmountB, Bytes(''), Bytes(''), false)
  }

  swap(input: gtxn.Transaction, poolId: uint64, minAmountOut: uint64, recipient: Account, data: bytes): uint64 {
    const escrow = this.escrowOf(this.loadPool(poolId))
    const deposit = this.readDeposit(input, escrow)
    const to = this.resolveRecipient(recipient)
    const result = this.swapCore(poolId, deposit.asset, deposit.amount, to, data)
    loggedAssert(result.amountOut >= minAmountOut, errSlippage)
    this.send(escrow, to, result.assetOut, result.amountOut)
    return result.amountOut
  }

  swapRoute(input: gtxn.Transaction, route: uint64[], minAmountOut: uint64, recipient: Account, data: bytes): uint64 {
    loggedAssert(route.length > 0 && route.length <= MaxRouteHops, errInvalidRoute)
    let escrow = this.escrowOf(this.loadPool(route[0]))
    const deposit = this.readDeposit(input, escrow)
    const to = this.resolveRecipient(recipient)
    let asset = deposit.asset
    let amount = deposit.amount
    for (let i: uint64 = 0; i < route.length; i += 1) {
      if (i > 0) ensureBudget(HopBudget, OpUpFeeSource.GroupCredit)
      const result = this.swapCore(route[i], asset, amount, to, data)
      asset = result.assetOut
      amount = result.amountOut
      if (i + 1 < route.length) {
        const next = this.escrowOf(this.loadPool(route[i + 1]))
        this.send(escrow, next, asset, amount)
        escrow = next
      }
    }
    loggedAssert(amount >= minAmountOut, errSlippage)
    this.send(escrow, to, asset, amount)
    return amount
  }

  @abimethod({ readonly: true })
  quote(poolId: uint64, assetIn: uint64, amountIn: uint64): uint64 {
    return this.quoteCore(poolId, assetIn, amountIn).amountOut
  }

  @abimethod({ readonly: true })
  quoteRoute(route: uint64[], assetIn: uint64, amountIn: uint64): uint64 {
    loggedAssert(route.length > 0 && route.length <= MaxRouteHops, errInvalidRoute)
    let asset = assetIn
    let amount = amountIn
    for (let i: uint64 = 0; i < route.length; i += 1) {
      if (i > 0) ensureBudget(HopBudget, OpUpFeeSource.GroupCredit)
      const result = this.quoteCore(route[i], asset, amount)
      asset = result.assetOut
      amount = result.amountOut
    }
    return amount
  }

  @abimethod({ readonly: true })
  getPool(poolId: uint64): PoolInfo {
    return this.loadPool(poolId)
  }

  @abimethod({ readonly: true })
  getPoolHooks(poolId: uint64): HookConfig[] {
    if (!this.poolHooks(poolId).exists) {
      return []
    }
    return clone(this.poolHooks(poolId).value)
  }

  @abimethod({ readonly: true })
  getPairPools(assetA: uint64, assetB: uint64): uint64[] {
    const key: PairKey = { assetA, assetB }
    if (!this.pairPools(key).exists) {
      return []
    }
    return clone(this.pairPools(key).value)
  }

  @abimethod({ readonly: true })
  getHookApproval(hook: uint64): HookApproval {
    let a: HookApproval = { perms: 0, extCapBps: 0 }
    if (this.approvedHooks(hook).exists) a = clone(this.approvedHooks(hook).value)
    return a
  }
