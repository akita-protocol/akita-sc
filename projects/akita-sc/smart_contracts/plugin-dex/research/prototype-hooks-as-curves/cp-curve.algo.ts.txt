import { Application, BigUint, Bytes, bytes, Contract, Global, GlobalState, loggedAssert, op, uint64 } from '@algorandfoundation/algorand-typescript'
import { abimethod } from '@algorandfoundation/algorand-typescript/arc4'
import { BpsDivisor, CurveOpAdd, CurveOpInit, CurveOpQuote, CurveOpRemove, CurveOpSettle, CurveOpSwap, MinimumLiquidity } from '../constants'
import { CurveRequest, CurveResult } from '../types'

/**
 * Stateless constant-product curve. Everything it needs is in the request (the DEX
 * ledger), so quote == swap by construction and there is nothing to keep in sync.
 */
export class CpCurve extends Contract {
  dex = GlobalState<Application>({ key: 'dex' })

  @abimethod({ onCreate: 'require' })
  create(dex: Application): void {
    this.dex.value = dex
  }

  private mulDiv(a: uint64, b: uint64, c: uint64): uint64 {
    return op.divw(...op.mulw(a, b), c)
  }

  private price(a: uint64, b: uint64): bytes<16> {
    if (a === 0) return op.bzero(16).toFixed({ length: 16 })
    return op
      .bzero(16)
      .bitwiseOr(Bytes((BigUint(b) * BigUint(Bytes.fromHex('010000000000000000'))) / BigUint(a)))
      .toFixed({ length: 16 })
  }

  akitaDexCurve(dex: Application, req: CurveRequest): CurveResult {
    loggedAssert(Global.callerApplicationId === this.dex.value.id, 'errCurveCaller')
    let rA = req.reserveA
    let rB = req.reserveB
    const res: CurveResult = { amountA: 0, amountB: 0, lpFee: 0, shares: 0, priceX64: this.price(rA, rB), activeId: 0, reserveA: rA, reserveB: rB, flags: 0 }

    if (req.op === CurveOpSwap || req.op === CurveOpQuote) {
      const amountIn: uint64 = req.inIsA ? req.amountA : req.amountB
      const rIn: uint64 = req.inIsA ? rA : rB
      const rOut: uint64 = req.inIsA ? rB : rA
      const lpFee = this.mulDiv(amountIn, req.feeBps, BpsDivisor)
      const out = this.mulDiv(amountIn - lpFee, rOut, rIn + amountIn - lpFee)
      const protocolFee = this.mulDiv(lpFee, req.protocolFeeBps, BpsDivisor)
      res.lpFee = lpFee
      if (req.inIsA) {
        res.amountA = amountIn
        res.amountB = out
        rA = rA + amountIn - protocolFee
        rB = rB - out
      } else {
        res.amountB = amountIn
        res.amountA = out
        rB = rB + amountIn - protocolFee
        rA = rA - out
      }
    } else if (req.op === CurveOpAdd) {
      if (req.lpSupply === 0) {
        const l = op.btoi(Bytes(op.bsqrt(BigUint(req.amountA) * BigUint(req.amountB))))
        loggedAssert(l > MinimumLiquidity, 'errLiq')
        res.shares = l
        res.amountA = req.amountA
        res.amountB = req.amountB
      } else {
        const fromA = this.mulDiv(req.amountA, req.lpSupply, rA)
        const fromB = this.mulDiv(req.amountB, req.lpSupply, rB)
        res.shares = fromA <= fromB ? fromA : fromB
        res.amountA = fromA <= fromB ? req.amountA : this.mulDiv(req.amountB, rA, rB) + 1
        res.amountB = fromA <= fromB ? this.mulDiv(req.amountA, rB, rA) + 1 : req.amountB
      }
      rA += res.amountA
      rB += res.amountB
    } else if (req.op === CurveOpRemove) {
      res.shares = req.lpAmount
      res.amountA = this.mulDiv(req.lpAmount, rA, req.lpSupply)
      res.amountB = this.mulDiv(req.lpAmount, rB, req.lpSupply)
      rA -= res.amountA
      rB -= res.amountB
    } else if (req.op === CurveOpSettle) {
      // custody delta: donation, or a trade that must not decrease k
      const nA: uint64 = req.negA ? rA - req.amountA : rA + req.amountA
      const nB: uint64 = req.negB ? rB - req.amountB : rB + req.amountB
      loggedAssert(BigUint(nA) * BigUint(nB) >= BigUint(rA) * BigUint(rB), 'errK')
      rA = nA
      rB = nB
    } else {
      loggedAssert(req.op === CurveOpInit, 'errOp')
    }
    res.reserveA = rA
    res.reserveB = rB
    res.priceX64 = this.price(rA, rB)
    return res
  }
}
