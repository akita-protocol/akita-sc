import { Application, BigUint, biguint, BoxMap, clone, Bytes, bytes, Contract, Global, GlobalState, loggedAssert, op, uint64 } from '@algorandfoundation/algorand-typescript'
import { abimethod } from '@algorandfoundation/algorand-typescript/arc4'
import { BpsDivisor, CurveOpQuote, CurveOpSwap } from '../constants'
import { CurveRequest, CurveResult } from '../types'

const BinsPerPage: uint64 = 32
const BinBytes: uint64 = 16 // x u64 | y u64
const PageBytes: uint64 = 512
const Q64 = 'q'

export type BinPool = {
  binStep: uint64
  activeId: uint64
  /** Q64.64 price of the active bin (B per A), cached so steps are one mulDiv */
  activePriceX64: bytes<16>
  reserveA: uint64
  reserveB: uint64
}

/** Swap-path sketch of a Liquidity Book style curve, used to size budget per crossed bin */
export class BinCurve extends Contract {
  dex = GlobalState<Application>({ key: 'dex' })
  pools = BoxMap<uint64, BinPool>({ keyPrefix: 'b' })

  @abimethod({ onCreate: 'require' })
  create(dex: Application): void {
    this.dex.value = dex
  }

  private pageKey(poolId: uint64, page: uint64): bytes {
    return Bytes('g').concat(op.itob(poolId)).concat(op.itob(page))
  }

  akitaDexCurve(dex: Application, req: CurveRequest): CurveResult {
    loggedAssert(Global.callerApplicationId === this.dex.value.id, 'errCurveCaller')
    const st = clone(this.pools(req.poolId).value)
    const commit = req.op === CurveOpSwap
    const one: biguint = BigUint(Bytes.fromHex('010000000000000000'))
    const stepMul: biguint = BigUint(BpsDivisor + st.binStep)
    const bps: biguint = BigUint(BpsDivisor)

    let remaining: uint64 = req.inIsA ? req.amountA : req.amountB
    let out: uint64 = 0
    let lpFee: uint64 = 0
    let id = st.activeId
    let price: biguint = BigUint(st.activePriceX64)
    let page: uint64 = id / BinsPerPage
    let key = this.pageKey(req.poolId, page)
    let buf = op.Box.get(key)[0]

    while (remaining > 0) {
      const slot: uint64 = (id % BinsPerPage) * BinBytes
      const x = op.extractUint64(buf, slot)
      const y = op.extractUint64(buf, slot + 8)
      const fee: uint64 = op.divw(...op.mulw(remaining, req.feeBps), BpsDivisor)
      const net: uint64 = remaining - fee
      if (req.inIsA) {
        // sell A: receive B at `price`
        const wantB: uint64 = op.btoi(Bytes((BigUint(net) * price) / one))
        const give: uint64 = wantB < y ? wantB : y
        const used: uint64 = give === wantB ? net : op.btoi(Bytes((BigUint(give) * one) / price))
        const usedFee: uint64 = give === wantB ? fee : op.divw(...op.mulw(used, req.feeBps), BpsDivisor - req.feeBps)
        buf = op.replace(buf, slot, op.itob(x + used + usedFee).concat(op.itob(y - give)))
        out += give
        lpFee += usedFee
        remaining -= used + usedFee
        if (remaining > 0) {
          id -= 1
          price = (price * bps) / stepMul
        }
      } else {
        const wantA: uint64 = op.btoi(Bytes((BigUint(net) * one) / price))
        const give: uint64 = wantA < x ? wantA : x
        const used: uint64 = give === wantA ? net : op.btoi(Bytes((BigUint(give) * price) / one))
        const usedFee: uint64 = give === wantA ? fee : op.divw(...op.mulw(used, req.feeBps), BpsDivisor - req.feeBps)
        buf = op.replace(buf, slot, op.itob(x - give).concat(op.itob(y + used + usedFee)))
        out += give
        lpFee += usedFee
        remaining -= used + usedFee
        if (remaining > 0) {
          id += 1
          price = (price * stepMul) / bps
        }
      }
      if (id / BinsPerPage !== page) {
        if (commit) op.Box.put(key, buf)
        page = id / BinsPerPage
        key = this.pageKey(req.poolId, page)
        buf = op.Box.get(key)[0]
      }
    }
    if (commit) op.Box.put(key, buf)

    const amountIn: uint64 = req.inIsA ? req.amountA : req.amountB
    const pf: uint64 = op.divw(...op.mulw(lpFee, req.protocolFeeBps), BpsDivisor)
    const rA: uint64 = req.inIsA ? st.reserveA + amountIn - pf : st.reserveA - out
    const rB: uint64 = req.inIsA ? st.reserveB - out : st.reserveB + amountIn - pf
    const priceBytes = op.bzero(16).bitwiseOr(Bytes(price)).toFixed({ length: 16 })
    if (commit) {
      this.pools(req.poolId).value = { binStep: st.binStep, activeId: id, activePriceX64: priceBytes, reserveA: rA, reserveB: rB }
    }
    return {
      amountA: req.inIsA ? amountIn : out,
      amountB: req.inIsA ? out : amountIn,
      lpFee,
      shares: 0,
      priceX64: priceBytes,
      activeId: id,
      reserveA: rA,
      reserveB: rB,
      flags: 0,
    }
  }
}
